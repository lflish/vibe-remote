package session

import (
	"path/filepath"
	"testing"
)

func TestV2InheritedCLocale(t *testing.T) {
	requireTmux(t)
	for _, dirName := range []string{"ascii", "项目"} {
		t.Run(dirName, func(t *testing.T) {
			t.Setenv("LC_ALL", "C")
			freshTmuxDir(t)
			repo := repoAt(t, filepath.Join(canonPath(t, t.TempDir()), dirName, "devflow"))
			root := externalWorktree(t, repo, "feat")
			store := NewMemoryProjectStore()
			m1 := NewManager(true, "sleep 120", false, "")
			m1.SetProjectStore(store)
			id := terminalSessionIn(t, m1, root)
			m2 := NewManager(true, "sleep 120", false, "")
			m2.SetProjectStore(store)
			live, ok := m2.liveTmuxSessions()
			t.Logf("restarted daemon: read ok=%v count=%d record=%+v", ok, len(live), live[id])
			st, err := m2.WorktreeStatus(root, allowAny)
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("restarted daemon: status.sessions=%v (session %s was created in this worktree)", st.Sessions, id)
			res, rerr := m2.RemoveWorktree(request(st), allowAny)
			alive := tmuxCmd("has-session", "-t", "vibe-remote-"+id).Run() == nil
			t.Logf("restarted daemon: removal=%+v err=%v directoryExists=%v tmuxSessionAlive=%v", res, rerr, pathExists(root), alive)
			if res.Removed && alive {
				t.Errorf("worktree removed while the session created in it is still alive (endedSessions=%v)", res.EndedSessions)
			}
		})
	}
}
